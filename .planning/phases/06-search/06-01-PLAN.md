    **Step 2: Add `searchAssets` to `frontend/src/lib/api.ts`** (append to existing file):

    ```typescript
    export async function searchAssets(q: string, tags?: string[]): Promise<SearchResponse> {
      const params = new URLSearchParams();
      params.append('q', q);
      if (tags && tags.length > 0) {
        tags.forEach(t => params.append('tags', t));
      }
      const res = await fetch(`${API_BASE}/search?${params.toString()}`);
      if (res.status === 503) {
        return { results: [], error: 'search_unavailable' };
      }
      if (!res.ok) throw new Error(`Search failed: ${res.status}`);
      return res.json();
    }
    ```

    Add `import type { SearchResponse }` to the existing imports from `'../types/asset'` at the top of api.ts.

    **Step 3: Create `frontend/src/hooks/useSearch.ts`**:

    ```typescript
    import { useQuery } from '@tanstack/react-query';
    import { searchAssets } from '../lib/api';
    import type { SearchResponse } from '../types/asset';

    export function useSearch(query: string, tags?: string[]) {
      return useQuery<SearchResponse>({
        queryKey: ['search', query, tags],
        queryFn: () => searchAssets(query, tags),
        enabled: query.trim().length > 0,  // Only query when there's actual text
        staleTime: 30_000,  // Match existing staleTime convention
      });
    }
    ```

    Key design: `enabled: query.trim().length > 0` means the hook does nothing when query is empty -- the grid shows all assets via the regular `useAssets` hook. This implements the "clearing search returns to full grid" behavior without any conditional logic in the UI.
