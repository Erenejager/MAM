   ```typescript
   import { FastifyInstance } from 'fastify';
   import multipart from '@fastify/multipart';

   import { pipelineQueue } from '../lib/queue.js';
   ```

   Register `@fastify/multipart` with `limits: { fileSize: 10 * 1024 * 1024 * 1024 }` (10 GB).

   **POST /api/assets:**
   - Call `request.file()` to get the multipart file stream
   - If no file: return 400 `{ error: 'No file uploaded' }`
   - Generate `assetId = randomUUID()`
   - Get STORAGE_ROOT from `process.env.STORAGE_ROOT`
   - Create asset directory: `STORAGE_ROOT/{assetId}/`
   - Determine original extension: `extname(file.filename)` or fallback to `.mp4`
   - Stream file to disk via `saveAndHash(file.file, resolve(storageRoot, assetId, 'original' + ext))`
   - Get `fileHash` from saveAndHash return value
   - Check for duplicate: `db.select({ id: assets.id }).from(assets).where(eq(assets.fileHash, fileHash)).get()`
   - If duplicate exists: `await rm(assetDir, { recursive: true, force: true })` and return 409 `{ error: 'This file has already been imported.', existingId: existing.id }`
   - If NOT duplicate: insert asset record into SQLite:
     ```typescript
     db.insert(assets).values({
       id: assetId,
       originalFilename: file.filename,
       filepath: `${assetId}/original${ext}`,  // relative to STORAGE_ROOT
       fileSize: /* get from stat or hash byte count */,
       fileHash,
       status: 'ingesting',
       title: file.filename.replace(/\.[^.]+$/, ''),  // filename without extension as default title
       metadataStatus: 'pending',
       thumbnailStatus: 'pending',
       transcriptionStatus: 'pending',
       searchIndexStatus: 'pending',
     }).run();
     ```
   - Return 202 `{ id: assetId, status: 'ingesting' }`
   - AFTER reply is sent: enqueue pipeline job via `pipelineQueue.add(() => runPipeline(assetId))` — the `runPipeline` import will be wired in Plan 02. For now, create a placeholder: `async function runPipeline(assetId: string) { /* wired in Plan 02 */ }`
   - Wrap the entire handler in try/catch — on error, clean up partial files: `await rm(assetDir, { recursive: true, force: true }).catch(() => {})` and return 500

   **GET /api/assets/:id:**
   - Extract `id` from `request.params`
   - Query: `db.select().from(assets).where(eq(assets.id, id)).get()`
   - If not found: return 404 `{ error: 'Asset not found' }`
   - Return the full asset record as JSON

   Export the plugin function as `assetRoutes`.
