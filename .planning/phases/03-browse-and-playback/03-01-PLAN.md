**1. GET /api/assets** (list with optional tag filter):
```typescript
fastify.get<{
  Querystring: { tags?: string | string[] }
}>('/api/assets', async (request, reply) => {
  const { tags } = request.query;
  const tagList = tags ? (Array.isArray(tags) ? tags : [tags]) : [];

  if (tagList.length > 0) {
    return db.select().from(assets)
      .where(sql`(SELECT COUNT(DISTINCT value) FROM json_each(${assets.tags}) WHERE value IN (${sql.join(tagList.map(t => sql`${t}`), sql`, `)})) = ${tagList.length}`)
      .orderBy(sql`${assets.createdAt} DESC`)
      .all();
  }
  return db.select().from(assets).orderBy(sql`${assets.createdAt} DESC`).all();
});
```

**2. GET /api/tags** (unique tags with counts):
```typescript
fastify.get('/api/tags', async () => {
  const rows = db.all<{ tag: string; count: number }>(
    sql`SELECT value as tag, COUNT(*) as count FROM assets, json_each(assets.tags) GROUP BY value ORDER BY value`
  );
  return rows;
});
```
Note: Use raw SQL via `db.run` or the drizzle `sql` tagged template since json_each requires raw SQL. If drizzle's `db.all()` doesn't exist, use the underlying `better-sqlite3` instance: `db.$client.prepare(...).all()` or `db.execute(sql...)`.

**3. DELETE /api/assets/:id**:
```typescript
fastify.delete<{
  Params: { id: string };
  Querystring: { deleteFile?: string };
}>('/api/assets/:id', async (request, reply) => {
  const { id } = request.params;
  const deleteFile = request.query.deleteFile === 'true';

  const asset = db.select().from(assets).where(eq(assets.id, id)).get();
  if (!asset) return reply.status(404).send({ error: 'Asset not found' });

  db.delete(assets).where(eq(assets.id, id)).run();

  if (deleteFile) {
    const storageRoot = process.env.STORAGE_ROOT!;
    const assetDir = resolve(storageRoot, id);
    await rm(assetDir, { recursive: true, force: true });
  }

  return reply.status(204).send();
});
```

**4. PATCH /api/assets/:id**:
```typescript
fastify.patch<{
  Params: { id: string };
  Body: { tags?: string[] };
}>('/api/assets/:id', async (request, reply) => {
  const { id } = request.params;
  const { tags } = request.body as { tags?: string[] };

  const asset = db.select().from(assets).where(eq(assets.id, id)).get();
  if (!asset) return reply.status(404).send({ error: 'Asset not found' });

  if (tags !== undefined) {
    db.update(assets)
      .set({ tags: JSON.stringify(tags), updatedAt: new Date().toISOString() })
      .where(eq(assets.id, id))
      .run();
  }

  return db.select().from(assets).where(eq(assets.id, id)).get();
});
```

Add these imports at the top: `import { sql } from 'drizzle-orm';` and `import { rm } from 'node:fs/promises';` (rm may already be imported).
